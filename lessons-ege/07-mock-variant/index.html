def read_graph(path: str) -> dict[int, list[tuple[int, float]]]:
    graph = {}
    for line in open(path):
        a, b, w = line.split()
        a, b, w = int(a), int(b), float(w)
        if a not in graph:
            graph[a] = []
        if b not in graph:
            graph[b] = []
        graph[a].append((b, w))
    return graph


# Длина кратчайшего пути из start в finish.
def shortest(graph: dict[int, list[tuple[int, float]]], start: int, finish: int) -> float:
    memo = {}

    def best(v):
        if v == finish:
            return 0
        if v in memo:
            return memo[v]
        res = float('inf')
        for to, w in graph[v]:
            res = min(res, w + best(to))
        memo[v] = res
        return res

    return best(start)


print(int(shortest(read_graph('23.txt'), 1, 100)))

def read_requests(path: str) -> list[tuple[int, int, int]]:
    lines = open(path).readlines()
    res = []
    for line in lines[1:]:
        req, house, entrance = line.split()
        res.append((int(req), int(house), int(entrance)))
    return res


def answer(requests: list[tuple[int, int, int]]) -> tuple[int, int]:
    # houses[дом][подъезд] = наибольший номер заявки в этом подъезде
    houses = {}
    for req, house, entrance in requests:
        if house not in houses:
            houses[house] = {}
        houses[house][entrance] = max(houses[house].get(entrance, 0), req)

    best_len, best_req, best_house, best_first = 0, 0, 0, 0
    for house in houses:
        entrances = sorted(houses[house])
        first = entrances[0]
        for i in range(len(entrances)):
            if i > 0 and entrances[i] != entrances[i - 1] + 1:
                first = entrances[i]
            length = entrances[i] - first + 1
            req = houses[house][first]
            if length > best_len or (length == best_len and req > best_req):
                best_len, best_req, best_house, best_first = length, req, house, first
    return best_house, best_first


print(answer(read_requests('26.txt')))

def dist(p: tuple[float, float], q: tuple[float, float]) -> float:
    return ((p[0] - q[0]) ** 2 + (p[1] - q[1]) ** 2) ** 0.5


# Кластеры видно на картинке: делим точки условиями на координаты, помехи не попадают никуда.
def clusters_a(pts: list[tuple[float, float]]) -> list[list[tuple[float, float]]]:
    left, right = [], []
    for x, y in pts:
        if x < 0:
            left.append((x, y))
        else:

# Межкластерный диаметр: (расстояние, точка из первого, точка из второго).
def diameter(c1: list[tuple[float, float]], c2: list[tuple[float, float]]) -> tuple[float, tuple, tuple]:
    best = (0, None, None)
    for p in c1:
        for q in c2:
            d = dist(p, q)
            if d > best[0]:
                best = (d, p, q)
    return best


def answer_a(pts: list[tuple[float, float]]) -> tuple[int, int]:
    c = clusters_a(pts)
    d, p, q = diameter(c[0], c[1])
    return int(abs(p[0] - q[0]) * 1000), int(abs((p[1] + q[1]) * 1000))


def answer_b(pts: list[tuple[float, float]]) -> tuple[int, int]:
    c = clusters_b(pts)
    q1 = 0
    q2 = 0
    for i in range(3):
        for j in range(i + 1, 3):
            d, p, q = diameter(c[i], c[j])
            q1 += d
            q2 = max(q2, dist(p, (1, 1)), dist(q, (1, 1)))
    return int(q1 * 100), int(q2 * 100)
